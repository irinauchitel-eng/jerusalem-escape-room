public class JewishQuarter
{
    public string SiteName { get; set; }
    public int    YearBuilt { get; set; }
    public int    Visitors  { get; set; }

    public JewishQuarter(string name, int year, int visitors)
    {
        SiteName  = name;
        YearBuilt = year;
        Visitors  = visitors;
    }

    public string GetInfo()
    {
        return SiteName + " - " + YearBuilt;
    }
}

נתונות שתי מחלקות:

public class Site
{
    public string Name     { get; set; }
    public int    Visitors { get; set; }
    public Site(string n, int v) { Name=n; Visitors=v; }
}

public class Quarter
{
    public List<Site> Sites { get; set; }
    public Quarter() { Sites = new List<Site>(); }

    public static int TotalVisitors(Quarter q)
    {
        int total = 0;
        for (int i = 0; i < q.Sites.Count; i++)
            total += q.Sites[i].Visitors;
        return total;
    }
}

מה יודפס?

Quarter arm = new Quarter();
arm.Sites.Add(new Site("מנזר סנט ג'יימס", 80000));
arm.Sites.Add(new Site("ארמון האבנים",    45000));
Console.WriteLine(Quarter.TotalVisitors(arm));
